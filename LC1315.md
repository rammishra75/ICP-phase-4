# Sum of Nodes of Even valued Granparents

```
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public int findval(TreeNode node){
        return node == null? 0 : node.val;
    }
    public int sumEvenGrandparent(TreeNode root) {
        if(root == null) return 0;
        Queue<TreeNode> q = new LinkedList<>();
        int sum = 0;
        q.add(root);
        while(!q.isEmpty()){
            TreeNode node = q.remove();
            if(node.val % 2 == 0){
                if(node.left != null){
                sum += findval(node.left.left) + findval(node.left.right);
                }
                if(node.right != null){
                    sum += findval(node.right.left) + findval(node.right.right);
                } 
            }
            if(node.left != null){
                q.add(node.left);
            }
            if(node.right != null){
                q.add(node.right);
            }
        }
        return sum;
    }
}
```
