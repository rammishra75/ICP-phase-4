# LC 3951. Minimum Energy to Maintain Brightness

```
class Solution {
    public long minEnergy(int n, int brightness, int[][] intervals) {
        long ans = 0;
        long need = 0;
        if(brightness <= 3) need = 1;
        else{
            need = brightness / 3;
            brightness %= 3;
            if(brightness > 0) need++;
        }
        Arrays.sort(intervals, (a, b) -> Integer.compare(a[0], b[0]));
        long interval = intervals[0][1] - intervals[0][0] + 1;
        int last_max = intervals[0][1];

        for(int i = 1; i < intervals.length; i++) {
            if(intervals[i][0] > last_max) {
            interval += intervals[i][1] - intervals[i][0] + 1;
            }else if(intervals[i][1] > last_max) {
                interval += intervals[i][1] - last_max;
            }
            last_max = Math.max(last_max, intervals[i][1]);
        }
        ans = need * interval;
        return ans;
    }
}
```
