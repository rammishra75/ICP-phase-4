# Password Strength

```
class Solution {
    public int passwordStrength(String password) {
        int sum = 0;
        int n = password.length();
        Set<Character> set = new HashSet<>();
        for(int i = 0; i < n; i++){
            char c = password.charAt(i);
            if(!set.contains(c)){
                int val = (int)c;
                if(val >= 97 && val <= 122) sum += 1;
                else if(val >= 65 && val <= 90) sum += 2;
                else if(val >= 48 && val <= 57) sum += 3;
                else sum += 5;
            }
            set.add(c);
        }
        return sum;
    }
}
```
